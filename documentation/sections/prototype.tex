%% GUIDE: describe the 6-stage flow (matching test-order-workflow.js):
%%
%%  Stage 1 — Authentication
%%    POST /auth/signup + POST /auth/login
%%    Receive { accessToken, refreshToken }; open WebSocket to /notify.
%%
%%  Stage 2 — Browse Inventory
%%    GET /items  → Items Service → wms.items.request → WMS Adapter
%%    → wms.items.response → Notify Service → WebSocket { event: items_response }
%%
%%  Stage 3 — Create Order
%%    POST /order { itemList, destination } (x-request-id header as idempotency key)
%%    → cms.order.create → CMS Adapter (SOAP CreateOrder) → order.created
%%    → Payment Service creates payment intent
%%    → notify.order.payment_intent_created → client receives { orderId, clientSecret }
%%
%%  Stage 4 — Confirm Order
%%    PATCH /order { orderId, status: "order_confirmed", data: { paymentToken } }
%%    → cms.order.update_status → CMS Adapter (SOAP UpdateOrderStatus)
%%    → order.status_updated → order.confirmed → WMS Adapter (SOAP reserve)
%%    → order.wms.reserved → payment.charge → Payment Service charges
%%    → order.payment.completed → notify.order.payment_completed → client

%%    → CMS GetNextPendingDelivery → ROS optimize-route
%%    → driver receives route via WebSocket

%%    → WMS delivery-status update → client notified
%%

\label{subsec:realtime_tracking}

\subsubsection{Architectural Description}
%% GUIDE (full pipeline):
%%  1. Any service publishes  notify.<userId>.<event>  to "swift_logistics" exchange.
%%  2. Notify Service (unique per-instance queue: notifier.\${HOSTNAME})
%%     receives ALL notify.* events.
%%  3. Looks up the userId in its in-process Map<userId, Set<WebSocket>>.
%%  4. Online user: sends JSON frame to all connected sockets for that userId.
%%  5. Offline + persist=1: inserts into pending_notifications (PostgreSQL).
%%  6. On next WebSocket connect: flushes pending rows in timestamp order.
%%  Horizontal scaling: each instance has its own unique queue -- RabbitMQ
%%  delivers each notify.* event to EVERY Notify Service instance (fanout);
%%  each instance serves only its own connected users.
%%  Source: source-code/services/notify-service/index.js

\subsection{Setup and Running Instructions}
\label{subsec:setup}

\begin{lstlisting}[language=bash, caption=Building and Running the Prototype]
# 1. Clone the repository
git clone https://github.com/Ranidu-Premathilaka/SwiftLogistics.git
cd SwiftLogistics

# 2. (Optional) create a .env to override secrets
echo "JWT_SECRET=change-me-please" > .env

# 3. Build images and start all 13 containers
docker compose up --build

# 4. Verify gateway
curl http://localhost:8081/health        # {"status":"ok"}

# 5. RabbitMQ management UI: http://localhost:15672  (guest/guest)

# 6. Run integration tests (Node.js 20+ required on host)
cd source-code && npm install
node tests/test-all.js

# 7. Tear down
docker compose down
\end{lstlisting}
