\subsection{Challenge 1: Heterogeneous Systems Integration}
\label{subsec:challenge1}

\subsubsection{Problem Description}
% CMS (SOAP/XML), ROS (REST/JSON), WMS (TCP/IP proprietary)
% Protocol and data format differences
% Integration complexity

\subsubsection{Solution Approach}
% Adapter pattern implementation
% Message translator pattern
% Protocol translation mechanisms
% Data format transformation

\begin{figure}[H]
    \centering
    \includegraphics[width=\textwidth]{figures/heterogeneous_integration.pdf}
    \caption{Heterogeneous Systems Integration Solution}
    \label{fig:hetero_integration}
\end{figure}

\subsubsection{Implementation Details}
% CMS Adapter: SOAP client libraries
% ROS Adapter: REST client with JSON parsing
% WMS Adapter: Custom TCP/IP socket implementation
% Unified internal message format

\subsubsection{Benefits Achieved}
% Loose coupling between systems
% Flexibility to add new systems
% Centralized translation logic

\subsection{Challenge 4: Transaction Management}
\label{subsec:challenge4}

\subsubsection{Problem Description}
% Distributed transaction across CMS, WMS, ROS
% Ensuring consistency
% Handling partial failures
% Recovery mechanisms

\subsubsection{Solution Approach}
% Saga pattern for distributed transactions
% Compensating transactions
% Event sourcing for audit trail
% Idempotent operations

\begin{figure}[H]
    \centering
    \includegraphics[width=\textwidth]{figures/transaction_management.pdf}
    \caption{Distributed Transaction Management with Saga Pattern}
    \label{fig:transaction_mgmt}
\end{figure}

\subsubsection{Implementation Details}
% Saga orchestrator service
% Compensation handlers for rollback
% Transaction state machine
% Step-by-step transaction flow:
%   1. Order received
%   2. CMS creates order record
%   3. WMS reserves package slot
%   4. ROS adds to route
%   5. All steps committed or compensated

\subsubsection{Recovery Scenarios}
% Scenario 1: CMS failure (no compensation needed)
% Scenario 2: WMS failure (compensate CMS)
% Scenario 3: ROS failure (compensate WMS and CMS)
% Scenario 4: Network partition handling

\subsubsection{Benefits Achieved}
% Eventual consistency
% No lost orders
% Graceful failure handling
% Audit trail for all transactions
